cmake_minimum_required(VERSION 3.20)
project(ephemeris LANGUAGES C CXX)
add_library(ephemeris STATIC receiver/signal.cpp receiver/navigation.cpp receiver/app.cpp)
target_include_directories(ephemeris PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(ephemeris PUBLIC cxx_std_17)
target_compile_options(ephemeris PRIVATE -Wall -Wextra -Wpedantic)
if(CMAKE_CROSSCOMPILING)
  return()
endif()

option(EPHEMERIS_SANITIZE "Instrument native receiver and tests" OFF)
if(EPHEMERIS_SANITIZE)
  target_compile_options(ephemeris PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(ephemeris PUBLIC -fsanitize=address,undefined)
endif()
add_executable(eph-replay tools/replay.cpp)
target_link_libraries(eph-replay PRIVATE ephemeris)
enable_testing()
foreach(test signal navigation app)
  add_executable(${test}_test tests/${test}_test.cpp)
  target_link_libraries(${test}_test PRIVATE ephemeris)
  target_compile_options(${test}_test PRIVATE -UNDEBUG)
  add_test(NAME ${test} COMMAND ${test}_test)
endforeach()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME independent_replay COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tests/test_replay.py $<TARGET_FILE:eph-replay>)
add_test(NAME campaign_scoring COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tests/test_campaign.py)
add_test(NAME bench_pattern COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/firmware/test_bench.py)
add_executable(geometry_probe tests/geometry_probe.cpp)
target_link_libraries(geometry_probe PRIVATE ephemeris)
add_test(NAME leo_observables COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tests/leo_observables.py $<TARGET_FILE:geometry_probe>)
