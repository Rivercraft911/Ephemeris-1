cmake_minimum_required(VERSION 3.20)
project(ephemeris LANGUAGES C CXX)
add_library(ephemeris STATIC receiver/coarse.cpp receiver/bitplane.cpp receiver/signal.cpp receiver/navigation.cpp receiver/app.cpp)
target_include_directories(ephemeris PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(ephemeris PUBLIC cxx_std_17)
option(EPHEMERIS_GROUND_SEARCH "Try terrestrial Doppler before the full cold search" OFF)
if(EPHEMERIS_GROUND_SEARCH)
  target_compile_definitions(ephemeris PUBLIC EPHEMERIS_GROUND_SEARCH=1)
endif()
option(EPHEMERIS_GROUND_TROPOSPHERE "Apply a standard neutral-atmosphere delay to terrestrial code observations" OFF)
if(EPHEMERIS_GROUND_TROPOSPHERE)
  target_sources(ephemeris PRIVATE receiver/troposphere.cpp)
  target_compile_definitions(ephemeris PUBLIC EPHEMERIS_GROUND_TROPOSPHERE=1)
endif()
option(EPHEMERIS_BITPLANE_TRACKING "Use word-parallel closed-loop code and carrier tracking" OFF)
option(EPHEMERIS_COARSE_TRACKING "Track half-chip one-bit integrated samples" OFF)
option(EPHEMERIS_DOPPLER_POSITION "Smooth valid 5 Hz positions with measured Doppler velocity" OFF)
if(EPHEMERIS_DOPPLER_POSITION)
  target_compile_definitions(ephemeris PUBLIC EPHEMERIS_DOPPLER_POSITION=1)
endif()
option(EPHEMERIS_VELOCITY_FIT "Fit the current ECEF velocity to five successive Doppler solutions" OFF)
if(EPHEMERIS_VELOCITY_FIT)
  target_compile_definitions(ephemeris PUBLIC EPHEMERIS_VELOCITY_FIT=1)
endif()
if(EPHEMERIS_COARSE_TRACKING AND EPHEMERIS_BITPLANE_TRACKING)
  message(FATAL_ERROR "Choose a single live tracking kernel")
endif()
if(EPHEMERIS_COARSE_TRACKING)
  target_compile_definitions(ephemeris PUBLIC EPHEMERIS_COARSE_TRACKING=1)
endif()
if(EPHEMERIS_BITPLANE_TRACKING)
  target_compile_definitions(ephemeris PUBLIC EPHEMERIS_BITPLANE_TRACKING=1)
  option(EPHEMERIS_PREFIX_SUMS "Score the word kernel from shared I/Q prefixes instead of popcount" OFF)
  if(EPHEMERIS_PREFIX_SUMS)
    target_compile_definitions(ephemeris PUBLIC EPHEMERIS_PREFIX_SUMS=1)
  endif()
endif()
target_compile_options(ephemeris PRIVATE -Wall -Wextra -Wpedantic)
if(CMAKE_CROSSCOMPILING)
  # Match the Pico SDK section layout so the linker can drop unused receiver
  # kernels and their tables from each individual firmware image.
  target_compile_options(ephemeris PRIVATE -ffunction-sections -fdata-sections)
  return()
endif()

option(EPHEMERIS_SANITIZE "Instrument native receiver and tests" OFF)
if(EPHEMERIS_SANITIZE)
  target_compile_options(ephemeris PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(ephemeris PUBLIC -fsanitize=address,undefined)
endif()
add_library(eph_sky_mixer SHARED tools/sky_mixer.cpp)
target_compile_features(eph_sky_mixer PRIVATE cxx_std_17)
set_target_properties(eph_sky_mixer PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
if(MSVC)
  target_compile_options(eph_sky_mixer PRIVATE /fp:strict)
else()
  target_compile_options(eph_sky_mixer PRIVATE -ffp-contract=off)
endif()

add_executable(eph-replay tools/replay.cpp)
target_link_libraries(eph-replay PRIVATE ephemeris)
enable_testing()
if(EPHEMERIS_GROUND_TROPOSPHERE)
  add_executable(troposphere_test tests/troposphere_test.cpp)
  target_link_libraries(troposphere_test PRIVATE ephemeris)
  target_compile_options(troposphere_test PRIVATE -UNDEBUG)
  add_test(NAME troposphere COMMAND troposphere_test)
endif()
foreach(test signal bitplane coarse navigation app thread)
  add_executable(${test}_test tests/${test}_test.cpp)
  target_link_libraries(${test}_test PRIVATE ephemeris)
  target_compile_options(${test}_test PRIVATE -UNDEBUG)
  add_test(NAME ${test} COMMAND ${test}_test)
endforeach()
find_package(Threads REQUIRED)
target_link_libraries(thread_test PRIVATE Threads::Threads)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME import_baseband COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tests/test_import_baseband.py)
add_test(NAME ground_fix COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tests/test_ground_fix.py)
add_test(NAME independent_replay COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tests/test_replay.py $<TARGET_FILE:eph-replay>)
add_test(NAME sky_mixer_reference COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tests/test_sky_mixer.py $<TARGET_FILE:eph_sky_mixer>)
add_test(NAME campaign_scoring COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tests/test_campaign.py)
add_test(NAME conducted_evidence COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tests/test_qualify.py)
add_test(NAME target_replay_transport COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/firmware/test_replay_bench.py)
add_test(NAME clocked_replay COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/firmware/test_clocked_replay.py)
add_test(NAME bench_pattern COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/firmware/test_bench.py)
add_executable(geometry_probe tests/geometry_probe.cpp)
target_link_libraries(geometry_probe PRIVATE ephemeris)
add_test(NAME leo_observables COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tests/leo_observables.py $<TARGET_FILE:geometry_probe>)
