cmake_minimum_required(VERSION 3.20)
set(PICO_BOARD pimoroni_pico_plus2_rp2350 CACHE STRING "Pico SDK board")
if(NOT PICO_SDK_PATH)
  set(PICO_SDK_PATH "$ENV{PICO_SDK_PATH}")
endif()
include(${PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(ephemeris_firmware LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 17)
pico_sdk_init()
option(EPHEMERIS_CAPTURE_FALLING "Sample data after the falling HOST_CLK edge" OFF)

add_library(ephemeris_io STATIC board.cpp capture.cpp usb.cpp)
target_include_directories(ephemeris_io PUBLIC ${CMAKE_CURRENT_LIST_DIR})
target_compile_definitions(ephemeris_io PRIVATE EPHEMERIS_CAPTURE_FALLING=$<BOOL:${EPHEMERIS_CAPTURE_FALLING}>)
target_link_libraries(ephemeris_io PUBLIC pico_stdlib hardware_pio hardware_dma hardware_spi pico_unique_id tinyusb_device)

add_executable(ephemeris_capture bench.cpp kernel_bench.cpp coarse_bench.cpp pattern_source.cpp)
target_link_libraries(ephemeris_capture PRIVATE ephemeris_io)
pico_add_extra_outputs(ephemeris_capture)

add_executable(ephemeris_pattern pattern.cpp pattern_source.cpp)
target_link_libraries(ephemeris_pattern PRIVATE pico_stdlib hardware_pio hardware_dma hardware_clocks)
pico_add_extra_outputs(ephemeris_pattern)

add_subdirectory(${CMAKE_CURRENT_LIST_DIR}/.. ${CMAKE_CURRENT_BINARY_DIR}/core)
target_link_libraries(ephemeris_capture PRIVATE ephemeris)
target_compile_options(ephemeris PRIVATE -fstack-usage)
function(ephemeris_receiver_image target source)
  add_executable(${target} ${source})
  target_compile_options(${target} PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-fstack-usage>)
  target_compile_definitions(${target} PRIVATE PICO_USE_STACK_GUARDS=1 PICO_STACK_SIZE=4096 PICO_CORE1_STACK_SIZE=0)
  target_link_libraries(${target} PRIVATE ephemeris_io ephemeris pico_multicore)
  pico_add_extra_outputs(${target})
endfunction()
ephemeris_receiver_image(ephemeris_pico main.cpp)
ephemeris_receiver_image(ephemeris_profile profile.cpp)
